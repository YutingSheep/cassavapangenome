Non-reference nodes analysis：
提取path的名字
odgi paths -i ${i} -L | grep 'AM560' > reference_${i}.txt
提出非ref的node名字
odgi paths -i ${i} --non-reference-ranges reference_${i}.txt > non_reference_${i}.bed
矫正染色体名字
sed -i -e 's/#1#/HA/g' -e 's/#2#/HB/g' non_reference_${i}.bed
筛选>50 bp，产生bed文件
awk '{print $0, $3 - $2}' non_reference_${i}.bed > non_reference_${i}_sorted.bed
#根据自定义脚本对bed文件进行拆分
获得了bed后，就是和EDTA的注释进行overlap
例如：
bedtools intersect -a BRA001554_hap1.bed -b BRA001554_hap1.mod.EDTA.TEanno.gff3 -wa -wb|cut -f4-20 > BRA.tmp.gff3
perl gff2bed.pl BRA.tmp.gff3 structural > BRA.tmp.bed
perl -nle 'my ($chr, $s, $e, $anno, $dir, $supfam)=(split)[0,1,2,3,8,12]; print "10000 0.001 0.001 0.001 $chr $s $e NA $dir $anno $supfam"' BRA.tmp.bed > BRA.tmp.out
perl buildSummary.pl BRA.tmp.out > BRA.tmp.out.stat
参考：https://github.com/oushujun/EDTA/issues/169

这里举例一个集群脚本
##slrum script
!/bin/bash

#SBATCH -J cal
#SBATCH --partition=amd
#SBATCH --nodes=1
#SBATCH --ntasks-per-node=2
#SBATCH -o %j.out
#SBATCH -e %j.err

cd /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/0.cal/test

# 读取 name.txt 文件并逐行处理
cat name.txt|while read sample; do
  echo $sample
  for i in {1..18}; do
    chrom=$(printf "chr%02d" $i)

    # Hap1 processing
    bedtools intersect -a /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/0.cal/${sample}HA${chrom}.bed -b /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/EDTA.gff3/${sample}_hap1.mod.EDTA.TEanno.gff3 -wa -wb > ${sample}HA${chrom}.tmp.gff3.tmp
    python /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/intersect.gff.py -i ${sample}HA${chrom}.tmp.gff3.tmp -o ${sample}HA${chrom}.tmp.gff3
    perl /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/gff2bed.pl ${sample}HA${chrom}.tmp.gff3 structural > ${sample}HA${chrom}.tmp.bed
    perl -nle 'my ($chr, $s, $e, $anno, $dir, $supfam)=(split)[0,1,2,3,8,12]; print "10000 0.001 0.001 0.001 $chr $s $e NA $dir $anno $supfam"' ${sample}HA${chrom}.tmp.bed > ${sample}HA${chrom}.tmp.out
    perl /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/buildSummary.pl ${sample}HA${chrom}.tmp.out > ${sample}HA${chrom}.tmp.STAT

    # Hap2 processing
    bedtools intersect -a /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/0.cal/${sample}HB${chrom}.bed -b /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/EDTA.gff3/${sample}_hap2.mod.EDTA.TEanno.gff3 -wa -wb > ${sample}HB${chrom}.tmp.gff3.tmp
    python /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/intersect.gff.py -i ${sample}HB${chrom}.tmp.gff3.tmp -o ${sample}HB${chrom}.tmp.gff3
    perl /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/gff2bed.pl ${sample}HB${chrom}.tmp.gff3 structural > ${sample}HB${chrom}.tmp.bed
    perl -nle 'my ($chr, $s, $e, $anno, $dir, $supfam)=(split)[0,1,2,3,8,12]; print "10000 0.001 0.001 0.001 $chr $s $e NA $dir $anno $supfam"' ${sample}HB${chrom}.tmp.bed > ${sample}HB${chrom}.tmp.out
    perl /home/wangnan/4.cassava/7.pggb.chrs/04.extract_non_reference_paths/50_bp_bed/buildSummary.pl ${sample}HB${chrom}.tmp.out > ${sample}HB${chrom}.tmp.STAT
  done
done
